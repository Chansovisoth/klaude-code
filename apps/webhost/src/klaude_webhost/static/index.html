      from dataclasses import dataclass

@dataclass
class Message:
    role: str
    content: str

def summarize(messages: list[Message]) -> dict[str, int]:
    counts = {}
    for message in messages:
        counts[message.role] = counts.get(message.role, 0) + 1
    return counts

sample = [Message("assistant", "Ready when you are.")]
print(summarize(sample))
